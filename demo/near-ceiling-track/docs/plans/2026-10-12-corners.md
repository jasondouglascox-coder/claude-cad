> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ] `) syntax for tracking.

**Goal:** Generate every printed part of the bowstring corners (inside corners and the alcove bridge corner), their assemblies, build plates, and a wall-marking sheet, with tests proving fit, gauge, car clearance and printability.

**Architecture:** Pure corner geometry (`corner_layout.py`: portals, spans, supports, layout sheet) and span-local geometry (`span_geom.py`) feed new part modules (`bow_deck`, `bow_frame`, `floor_beam`, `rod_anchor`, `deep_bracket`, `rod_jig`). `corner_assembly.py` places everything plus the two aluminum rods in a corner frame for clash checks. `plates.corner_kit/corner_plates` pack corner kits; `build corners` writes STEP files and `corners_layout.csv`. The viaduct moves to a 1.75" wall offset (spec decision) and the bracket gains reusable `profile(yd1=...)` and `flanged()` helpers.

**Tech Stack:** Python 3.12 (uv), build123d 0.13.0, pytest.

**Spec:** `docs/specs/2026-10-12-corners-design.md` (and `docs/specs/2026-10-12-arch-deck-shelf-design.md` for the viaduct).

## Global Constraints

- Units are mm. Every dimension lives in `hotrain/params.py`.
- Corner frame: track enters along +X on y = 0 at P0, turns left 90° (center at +Y), and leaves along +Y. Walls are on the outside of the turn: wall A at y = −track_offset; wall B (inside corners) at x = curve end x + track_offset.
- Span frame: origin at portal A, x along chord A→B, y left (toward the curve center), z = 0 at the tie underside.
- `track_offset` = 1.75" (44.45 mm). `curve_r` derives from `alcove_w` (550.3 mm). `bow_span` = 2·R·sin 9° (172.2 mm).
- Every `make()` returns one valid solid, flat on Z = 0, within 251 mm, that also meshes for 3MF.
- Non-rectangular 2D outlines go through `joints.poly()` (counter-clockwise).
- Unions must overlap. Edge-to-edge unions make sliver triangles that invalidate the 3MF mesh (see `deep_bracket`).
- Commit after every task with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus

1. **A 60 ft car on the curve.** The body must clear the inner and outer arches at every position. `test_60ft_car_clears_frames`.
2. **Gauge on the curve.** Tie plates must set 16.6 mm gauge along the true arc, not the chord. `test_curve_deck_gauge[True]`.
3. **Corner meeting the viaduct.** The last viaduct bay's pier half must seat in the entry anchor's slot without clashing. `test_viaduct_pier_seats_in_entry_anchor`.
4. **Ceiling cables.** They must run from the beam eyes to the ceiling clear of the arches. `test_no_clashes_including_cables[bridge]`.
5. **Bambu Studio import.** Every corner part must produce a valid 3MF mesh. `test_every_corner_part_meshes_for_3mf`.

```python
        return self.pier_depth + self.bracket_bar

    def bed_usable(self) -> float:
        return self.bed - self.bed_margin
```

```python
"""Corner geometry. Corner frame: track enters along +X on y = 0 at portal P0, turns left (toward +Y)
around a curve of radius R, and leaves along +Y. The walls are on the outside (right) of the turn:
wall A at y = -track_offset; for an inside corner wall B is at x = curve end x + track_offset."""
import math
from dataclasses import dataclass

from hotrain.params import P, Params


@dataclass(frozen=True)
class Pose:
    x: float
    y: float
    heading: float  # degrees, 0 = +X


@dataclass(frozen=True)
class Span:
    index: int
    a: Pose
    b: Pose
    curved: bool


@dataclass(frozen=True)
class Corner:
    name: str
    n_in: int
    n_out: int
    wall_b: bool  # True for an inside corner (wall along the exit straights)
    ceiling: tuple  # portal indices hung from the ceiling
    deep: tuple  # (portal index, "A" | "B") deep wall brackets


def inside_corner() -> Corner:
    return Corner("inside", 1, 1, True, (), ((3, "A"), (4, "B")))


def bridge_corner() -> Corner:
    return Corner("bridge", 1, 3, False, (6,), ((3, "A"),))


def step_deg(p: Params) -> float:
    return 90 / p.curve_spans


def curve_start_x(p: Params, c: Corner) -> float:
    return c.n_in * p.bow_span


def portals(p: Params, c: Corner) -> list[Pose]:
    r, s, x1 = p.curve_r, p.bow_span, curve_start_x(p, c)
    out = [Pose(i * s, 0.0, 0.0) for i in range(c.n_in + 1)]
    for k in range(1, p.curve_spans + 1):
        a = math.radians(k * step_deg(p))
        out.append(Pose(x1 + r * math.sin(a), r - r * math.cos(a), k * step_deg(p)))
    xe, ye = x1 + r, r
    out += [Pose(xe, ye + j * s, 90.0) for j in range(1, c.n_out + 1)]
    return out


def spans(p: Params, c: Corner) -> list[Span]:
    ps = portals(p, c)
    return [Span(i, ps[i], ps[i + 1], c.n_in <= i < c.n_in + p.curve_spans) for i in range(len(ps) - 1)]


def track_point(p: Params, c: Corner, t: float) -> Pose:
    """Pose on the true track centerline at arc length t from P0."""
    r, x1 = p.curve_r, curve_start_x(p, c)
    if t <= x1:
        return Pose(t, 0.0, 0.0)
    arc = r * math.pi / 2
    if t <= x1 + arc:
        a = (t - x1) / r
        return Pose(x1 + r * math.sin(a), r - r * math.cos(a), math.degrees(a))
    return Pose(x1 + r, r + (t - x1 - arc), 90.0)


def path_length(p: Params, c: Corner) -> float:
    return curve_start_x(p, c) + p.curve_r * math.pi / 2 + c.n_out * p.bow_span


def chord_sagitta(p: Params) -> float:
    return p.bow_span ** 2 / (8 * p.curve_r)


def wall_offset(p: Params, c: Corner, pose: Pose) -> float:
    """Distance from the nearest real wall to a point on the track."""
    d = pose.y + p.track_offset
    if c.wall_b:
        d = min(d, curve_start_x(p, c) + p.curve_r + p.track_offset - pose.x)
    return d


def rod_length(p: Params, c: Corner, side: int) -> float:
    """Rod length between the end anchors (side +1 = inner rod, toward the curve center)."""
    straight = (c.n_in + c.n_out) * p.bow_span
    return straight + (p.curve_r - side * p.rod_y) * math.pi / 2


def layout_sheet(p: Params, c: Corner) -> list[tuple[str, str, float]]:
    """(item, how to measure, mm) for marking supports on the walls and ceiling.
    Inside corner: distances from the corner where the walls meet. Bridge corner: along the top wall
    from the bump-out face line; the ceiling drop also gives its distance out from the top wall."""
    ps = portals(p, c)
    xb = curve_start_x(p, c) + p.curve_r + p.track_offset
```

```python
"""Geometry inside one bowstring span, in the span frame: origin at portal A, x along the chord A->B,
y to the left (toward the curve center), z up with 0 at the tie underside."""
import math
from dataclasses import dataclass

from hotrain.params import Params


@dataclass(frozen=True)
class SpanGeom:
    s: float  # chord length
    curved: bool
    r: float  # track radius (curved spans)
    h: float  # curve center height above the chord: center at (s/2, h)
    alpha: float  # span angle, radians (0 for straight)
    yc: float  # frame center line: centered on the track's stray from the chord
    d: float  # each end stops this far short of the portal (room for the floor beam)


def span_geom(p: Params, curved: bool) -> SpanGeom:
    s, r = p.bow_span, p.curve_r
    d = p.fb_t / 2 + p.portal_gap
    if not curved:
        return SpanGeom(s, False, r, 0.0, 0.0, 0.0, d)
    h = math.sqrt(r * r - s * s / 4)
    return SpanGeom(s, True, r, h, 2 * math.asin(s / 2 / r), -(r - h) / 2, d)


def track_at(g: SpanGeom, f: float) -> tuple[float, float, float]:
    """(x, y, heading rad) on the track at fraction f in [0, 1] of the span."""
    if not g.curved:
        return f * g.s, 0.0, 0.0
    t = -g.alpha / 2 + f * g.alpha
    return g.s / 2 + g.r * math.sin(t), g.h - g.r * math.cos(t), t


def offset_at(g: SpanGeom, f: float, u: float) -> tuple[float, float]:
    """Point u to the left of the track (toward the curve center) at fraction f."""
    x, y, a = track_at(g, f)
    return x - u * math.sin(a), y + u * math.cos(a)


def track_len(g: SpanGeom) -> float:
    return g.r * g.alpha if g.curved else g.s


def end_fractions(g: SpanGeom) -> tuple[float, float]:
    """Fractions where the printed span starts/ends (short of the portals by d)."""
    e = g.d / track_len(g)
    return e, 1 - e


def frame_x_range(g: SpanGeom, y_line: float) -> tuple[float, float]:
    """Where the line y = y_line meets the span's (shortened) end lines."""
    if not g.curved:
        return g.d, g.s - g.d
    f0, f1 = end_fractions(g)
    out = []
    for f in (f0, f1):  # radial line from the center through the track point at f
        x, y, _ = track_at(g, f)
        cx, cy = g.s / 2, g.h
        k = (y_line - cy) / (y - cy)
        out.append(cx + k * (x - cx))
    return out[0], out[1]
```

```python
"""Wall bracket at each pier: flange on the stud, triangulated arm, deck block with pier slots and a latch."""
from build123d import (Location, Part, Plane, Pos, Rot, Sketch, SlotCenterToCenter, Box, Align,
                       extrude)

from hotrain.joints import poly, rect
from hotrain.params import P, Params


def slot_w(p: Params) -> float:
    return p.panel_t + p.fit_slide


def slots_and_latch(p: Params, slot_bottom: float, ys: tuple) -> tuple[list, Sketch]:
    """Pier slots at each y in ys (open at the top) plus the latch on the room-side slot.
    Returns (cuts, hook). Coordinates are (y, z) with z = 0 at the tie underside."""
    sw = slot_w(p)
    cuts = [rect(y - sw / 2, y + sw / 2, slot_bottom, 1) for y in ys]
    wall = max(ys) + sw / 2  # room-side wall of the room-side slot
    relief = wall + p.latch_finger_w
    cuts.append(rect(relief, relief + p.latch_relief, -p.latch_finger_len, 1))
    zb = -p.latch_top - p.latch_win_h + 0.5
    e = p.latch_engage
    hook = poly((wall + 0.05, zb), (wall - e, zb), (wall - e, zb + 0.8), (wall + 0.05, zb + 0.8 + e))
    return cuts, hook


def profile(p: Params, yd1: float | None = None) -> Sketch:
    """Profile in (y, z): y from the track centerline (+ room), z = 0 at the tie underside.
    yd1 overrides how far the deck block reaches toward the room (corner rod anchors reach further)."""
    yw = -p.track_offset
    yf = yw + p.flange_t
    yd0 = min(p.rail_ys) - slot_w(p) / 2 - p.deck_block_margin
    yd1 = p.tie_l / 2 if yd1 is None else yd1
    zb, m = -p.bracket_depth, p.bracket_member
    bound = rect(yw, yd1, zb, 0)
    p0, p1 = (yf, zb), (yd0, -m)
    dx, dy = p1[0] - p0[0], p1[1] - p0[1]
    ln = (dx * dx + dy * dy) ** 0.5
    nx, ny = -dy / ln * m / 2, dx / ln * m / 2
    diag = poly((p0[0] + nx, p0[1] + ny), (p1[0] + nx, p1[1] + ny), (p1[0] - nx, p1[1] - ny),
                (p0[0] - nx, p0[1] - ny)) & bound
    bus = rect(yf + 0.5, yf + 0.5 + p.bus_notch_w, p.bus_floor - 1, 1)  # wire pass-through
    if yd0 - yf < p.bracket_open_min:  # too short to triangulate: solid web
        body = rect(yw, yd1, zb, 0)
        cuts, hook = slots_and_latch(p, -p.pier_depth + p.pier_notch, p.rail_ys)
        return (body - cuts - bus) + hook
    body = Sketch() + [
        rect(yw, yf, zb, 0),  # wall plate
        rect(yw, yd0, -m, 0),  # top arm
        rect(yw, yd0, zb, zb + m),  # bottom arm
        diag,
        rect(yd0, yd1, zb, 0),  # deck block
    ]
    cuts, hook = slots_and_latch(p, -p.pier_depth + p.pier_notch, p.rail_ys)
    return (body - cuts - bus) + hook


def flanged(p: Params, prof: Sketch, depth: float, screw_zs: tuple) -> Part:
    """Extrude a bracket profile already shifted so the wall is at x = 0 and the bottom at y = 0,
    and add the wall flange with vertical screw slots (screw_zs measured down from the top)."""
    plate = extrude(prof, p.bracket_t)
    flange = Box(p.flange_t, depth, p.flange_w, align=(Align.MIN, Align.MIN, Align.MIN))
    zc = (p.bracket_t + p.flange_w) / 2  # centered on the exposed tab; this line goes on the stud center
    holes = [extrude(Plane.YZ * Pos(depth - z, zc) * SlotCenterToCenter(2 * p.screw_travel, p.screw_d),
                     p.flange_t) for z in screw_zs]
    return plate + flange - holes


def make(p: Params = P) -> Part:
    """Print orientation: plate flat (Z 0..bracket_t), flange standing up along the wall edge (X 0..flange_t)."""
    return flanged(p, Pos(p.track_offset, p.bracket_depth) * profile(p), p.bracket_depth, p.screw_zs)


def place(p: Params, xp: float) -> Location:
    return Pos(xp, 0, 0) * Rot(0, 0, 90) * Rot(90, 0, 0) * Pos(-p.track_offset, -p.bracket_depth, -p.bracket_t / 2)
```

```python
"""Bowstring side frame: bottom chord, arch rib, vertical hangers, end posts, tabs into the deck edge beam."""
from build123d import Bezier, Face, Line, Part, Pos, Sketch, Wire, extrude

from hotrain.joints import rect
from hotrain.params import P, Params


def tab_xs(p: Params, length: float) -> list[float]:
    return [length / 4, length / 2, 3 * length / 4]


def profile(p: Params, length: float) -> Sketch:
    """(x, z) with z = 0 at the tie underside; the frame stands on the edge beam top (z = tie_h)."""
    z0 = p.tie_h
    zc = z0 + p.bow_chord_h
    zs = zc + 4  # springing
    zt = p.rail_top + p.bow_rise  # crown, top of rib
    pw, rw = p.bow_post_w, p.bow_rib_w
    lo_mid = 2 * (zt - rw) - zs  # quadratic Bezier control so the curve peaks at zt - rw
    lo = Bezier((pw, zs), (length / 2, lo_mid), (length - pw, zs))
    hi = Bezier((length - pw, zs + rw), (length / 2, lo_mid + rw), (pw, zs + rw))  # lower edge + rw
    rib = Face(Wire([lo, Line((length - pw, zs), (length - pw, zs + rw)), hi, Line((pw, zs + rw), (pw, zs))]))
    shapes = [rect(0, length, z0, zc), rect(0, pw, z0, zs + rw), rect(length - pw, length, z0, zs + rw), rib]
    n = p.bow_panels
    for k in range(1, n):
        x = pw + k * (length - 2 * pw) / n
        t = (x - pw) / (length - 2 * pw)
        z_rib = (1 - t) ** 2 * zs + 2 * t * (1 - t) * lo_mid + t * t * zs
        shapes.append(rect(x - p.bow_hanger_w / 2, x + p.bow_hanger_w / 2, zc - 0.5, z_rib + 0.5))
    for x in tab_xs(p, length):
        shapes.append(rect(x - p.bow_tab_len / 2, x + p.bow_tab_len / 2, 0, z0 + 0.5))
    return Sketch() + shapes


def make(p: Params = P, length: float = 160.0) -> Part:
    """Print orientation: flat, x along the frame, Y = height above the tie underside."""
    return extrude(profile(p, length), p.bow_frame_t)
```

```python
"""Bowstring deck for one span: open ties on two edge beams, two rod runners with snap-down channels
following the true rod arcs, gauge-setting tie plates following the true track curve."""
import math

from build123d import Axis, Circle, Part, Plane, Polygon, Pos, Rectangle, Rot, Sketch, extrude, revolve

from hotrain.joints import poly, rect
from hotrain.params import P, Params
from hotrain.parts import bow_frame
from hotrain.span_geom import SpanGeom, end_fractions, frame_x_range, offset_at, span_geom, track_at, track_len


def _clip(g: SpanGeom) -> Sketch:
    """Region between the span's two (shortened) end lines."""
    if not g.curved:
        return rect(g.d, g.s - g.d, -200, 200)
    f0, f1 = end_fractions(g)
    cx, cy = g.s / 2, g.h
    pts = [(cx, cy)]
    for i in range(13):
        f = f0 + (f1 - f0) * i / 12
        x, y, _ = track_at(g, f)
        k = 1.6  # well past the outer edge beam
        pts.append((cx + k * (x - cx), cy + k * (y - cy)))
    return poly(*pts)


def _band(g: SpanGeom, p: Params) -> Sketch:
    lim = p.bow_frame_y + p.bow_edge_w / 2
    return rect(-50, g.s + 50, g.yc - lim, g.yc + lim)


def _oriented_rect(cx: float, cy: float, a: float, along: float, across: float, off_u: float = 0.0) -> Sketch:
    """Rectangle centered u = off_u to the left of (cx, cy), long side 'across' the track direction a."""
    return Pos(cx - off_u * math.sin(a), cy + off_u * math.cos(a)) * Rot(0, 0, math.degrees(a)) * Rectangle(along, across)


def _runner(p: Params, g: SpanGeom, u: float) -> Part:
    """Rod runner with a snap-down channel, swept along the rod's true path (u left of the track)."""
    w = p.rod_d / 2 + p.rod_wall
    prof_xz = Pos(0, 0) * Rectangle(2 * w, -p.runner_bot, align=None)
    f0, f1 = end_fractions(g)
    if not g.curved:
        face = Plane.YZ * (Pos(u, p.runner_bot / 2) * Rectangle(2 * w, -p.runner_bot) -
                           Pos(u, p.rod_zc) * Circle((p.rod_d + p.fit_slide) / 2))
        return Pos(g.d, 0, 0) * extrude(face, g.s - 2 * g.d)
    rr = g.r - u  # rod radius about the curve center
    face = Plane.XZ * (Pos(rr, p.runner_bot / 2) * Rectangle(2 * w, -p.runner_bot) -
                       Pos(rr, p.rod_zc) * Circle((p.rod_d + p.fit_slide) / 2))
    t0 = -g.alpha / 2 + f0 * g.alpha
    sweep = math.degrees((f1 - f0) * g.alpha)
    part = revolve(face, Axis.Z, sweep)
    start = math.degrees(t0) - 90  # polar angle about the center of the track point at f0
    return Pos(g.s / 2, g.h, 0) * Rot(0, 0, start) * part


def tie_fracs(p: Params, g: SpanGeom) -> list[float]:
    f0, f1 = end_fractions(g)
    n = max(1, round((f1 - f0) * track_len(g) / p.tie_pitch))
    return [f0 + (f1 - f0) * (k + 0.5) / n for k in range(n)]


def build(p: Params, curved: bool) -> Part:
    """Deck in the span frame."""
    g = span_geom(p, curved)
    clip, band = _clip(g), _band(g, p)
    across = 2 * (p.bow_frame_y + p.bow_edge_w)
    base = []
    for f in tie_fracs(p, g):
        x, y, a = track_at(g, f)
        base.append(_oriented_rect(x, y, a, p.tie_w, across))
    edges, slots = [], []
    for sgn in (1, -1):
        yl = g.yc + sgn * p.bow_frame_y
        edges.append(rect(-50, g.s + 50, yl - p.bow_edge_w / 2, yl + p.bow_edge_w / 2))
        xa, xb = frame_x_range(g, yl)
        ln = xb - xa - 2 * p.portal_gap
        for tx in bow_frame.tab_xs(p, ln):
            x = xa + p.portal_gap + tx
            slots.append(rect(x - (p.bow_tab_len + p.fit_slide) / 2, x + (p.bow_tab_len + p.fit_slide) / 2,
                              yl - (p.bow_frame_t + p.fit_slide) / 2, yl + (p.bow_frame_t + p.fit_slide) / 2))
    ties = ((Sketch() + base) & band & clip) - slots
    deck = extrude(ties, p.tie_h)
    edge = extrude(((Sketch() + edges) & clip) - slots, p.tie_h - p.runner_bot)
    deck += Pos(0, 0, p.runner_bot) * edge
    for u in (p.rod_y, -p.rod_y):
        deck += _runner(p, g, u)
    # tie plates, shoulders, outer spike lips along the true rails
    gw, sw = p.groove_w / 2, p.shoulder_w
    plates, shoulders, lips = [], [], []
    for k, f in enumerate(tie_fracs(p, g)):
        x, y, a = track_at(g, f)
        for rail in (p.rail_ctc / 2, -p.rail_ctc / 2):
            plates.append(_oriented_rect(x, y, a, p.tie_w, 2 * (gw + sw), rail))
            for side in (1, -1):
                shoulders.append(_oriented_rect(x, y, a, p.tie_w, sw, rail + side * (gw + sw / 2)))
            if k % p.spike_lip_every == 0:
                side = 1 if rail > 0 else -1  # outside of the rail only
                lips.append(_oriented_rect(x, y, a, p.tie_w, sw + p.spike_lip_overhang,
                                           rail + side * (gw + sw / 2 - p.spike_lip_overhang / 2)))
    z = p.tie_h
    deck += Pos(0, 0, z) * extrude(Sketch() + plates, p.tie_plate_h)
    z += p.tie_plate_h
    deck += Pos(0, 0, z) * extrude(Sketch() + shoulders, p.shoulder_h)
    z += p.shoulder_h
    if lips:
        deck += Pos(0, 0, z) * extrude(Sketch() + lips, p.spike_lip_t)
    return deck


def make(p: Params = P, curved: bool = True) -> Part:
    """Print orientation: runner mouths and edge beams on the bed, ties up."""
    d = build(p, curved)
    bb = d.bounding_box()
    return Pos(-bb.min.X, -bb.min.Y, -bb.min.Z) * d


def print_offset(p: Params, curved: bool) -> tuple[float, float, float]:
    bb = build(p, curved).bounding_box()
    return bb.min.X, bb.min.Y, bb.min.Z
```

```python
"""Floor beam under each portal between two bowstring spans. Threads onto both rods (closed bores),
carries cable eyes at its outer ends (used at a ceiling drop) and a side-entry bus-wire hook."""
from build123d import Circle, Location, Part, Pos, Rot, Sketch, extrude

from hotrain.corner_layout import chord_sagitta
from hotrain.joints import rect
from hotrain.params import P, Params


def half_len(p: Params) -> float:
    return p.bow_frame_y + chord_sagitta(p) + p.bow_edge_w / 2 + p.fb_reach


def bottom(p: Params) -> float:
    return p.runner_bot - p.fb_drop


def bus_u(p: Params) -> float:
    """Bus hook position, inner side, clear of where a deep bracket meets the beam's middle."""
    return p.rod_y


def profile(p: Params) -> Sketch:
    """(u, z): u left of the track (toward the curve center), z = 0 at the tie underside."""
    u, zb = half_len(p), bottom(p)
    body = rect(-u, u, zb, 0)
    hb = bus_u(p)
    hook = [rect(hb - 5, hb + 5, zb - 6, zb - 4.5), rect(hb - 5, hb - 3.5, zb - 6, zb + 0.5),
            rect(hb + 3.5, hb + 5, zb - 6, zb - 3)]
    holes = [Pos(y, p.rod_zc) * Circle((p.rod_d + p.fit_slide) / 2) for y in (p.rod_y, -p.rod_y)]
    holes += [Pos(y, (zb + 0) / 2) * Circle((p.cable_d + 0.5) / 2) for y in (u - 4, -(u - 4))]
    return (body + hook) - holes


def make(p: Params = P) -> Part:
    u, zb = half_len(p), bottom(p)
    return Pos(u, -(zb - 6), 0) * extrude(profile(p), p.fb_t)


def place(p: Params, x: float, y: float, heading_deg: float) -> Location:
    """Centered on the portal point, beam across the track, thickness along the heading."""
    u, zb = half_len(p), bottom(p)
    return (Pos(x, y, 0) * Rot(0, 0, heading_deg) * Rot(0, 0, 90) * Rot(90, 0, 0)
            * Pos(-u, zb - 6, -p.fb_t / 2))
```

```python
import math

from build123d import Location, Part, Pos, Rot, Sketch

from hotrain.joints import poly, rect
from hotrain.params import P, Params

DEPTH = 50.0


def top(p: Params) -> float:
    return floor_beam.bottom(p)


def band(p: Params, beam_angle_deg: float) -> float:
    """Half-width of the floor beam's footprint where it crosses the bracket's thickness."""
    a = math.radians(beam_angle_deg)
    return (p.fb_t / 2) / math.sin(a) + (p.bracket_t / 2) / math.tan(a)


def profile(p: Params, w_beam: float, beam_angle_deg: float) -> Sketch:
    """(w, z): w out from the wall, z from the bottom; the beam crosses at w = w_beam."""
    bw = band(p, beam_angle_deg) + p.portal_gap
    end = w_beam + bw + 3 + p.deep_reach_margin
    m, f = p.bracket_member, p.flange_t
    body = Sketch() + [
        rect(0, f, 0, DEPTH),
        rect(0, end, DEPTH - m, DEPTH),
        poly((f - 1, 0), (f + m * 1.4, 0), (end - 10, DEPTH - m + 1), (end - 10 - m * 1.4, DEPTH - m + 1)),  # overlaps, no slivers
        rect(w_beam - bw - 3, w_beam - bw, DEPTH, DEPTH + 3),  # stops either side of the beam
        rect(w_beam + bw, w_beam + bw + 3, DEPTH, DEPTH + 3),
    ]
    return body


def make(p: Params, w_beam: float, beam_angle_deg: float) -> Part:
    return bracket.flanged(p, profile(p, w_beam, beam_angle_deg), DEPTH, (DEPTH / 2,))


def place(p: Params, x_plane: float) -> Location:
```

```python
"""Bending jig: two stacked convex arcs. Bend the outer rod against the bottom step and the inner rod
against the top step, sliding the rod along as you go."""
import math

from build123d import Part, Pos, Sketch, extrude

from hotrain.joints import poly
from hotrain.params import P, Params

STEP_H = 8.0
WIDTH = 25.0


def _sector(r_in: float, r_out: float, half_angle: float) -> Sketch:
    pts = []
    for i in range(25):
        a = -half_angle + 2 * half_angle * i / 24
        pts.append((r_out * math.sin(a), r_out * math.cos(a)))
    for i in range(25):
        a = half_angle - 2 * half_angle * i / 24
        pts.append((r_in * math.sin(a), r_in * math.cos(a)))
    return poly(*pts)


def make(p: Params = P) -> Part:
    r_outer = p.curve_r + p.rod_y - p.rod_d / 2  # rod inner surface radius
    r_inner = p.curve_r - p.rod_y - p.rod_d / 2
    chord = p.bed_usable - 6
    half = math.asin(chord / 2 / r_outer)
    low = extrude(_sector(r_inner - WIDTH, r_outer, half), STEP_H)
    high = Pos(0, 0, STEP_H) * extrude(_sector(r_inner - WIDTH, r_inner, half), STEP_H)
    j = low + high
    bb = j.bounding_box()
    return Pos(-bb.min.X, -bb.min.Y, 0) * j
```

```python
"""Every part of one corner placed in the corner frame (see corner_layout), plus the rods,
for review and clash checks."""
import math

from build123d import Axis, Circle, Cylinder, Location, Part, Plane, Pos, Rot, Compound, extrude, revolve

from hotrain import corner_layout as C
from hotrain.params import P, Params
from hotrain.parts import bow_deck, bow_frame, deep_bracket, floor_beam, rod_anchor
from hotrain.span_geom import frame_x_range, span_geom


def wall_b_frame(p: Params, c: C.Corner) -> Location:
    """Maps a wall-A style frame (track along +x, wall at y = -o) onto the exit straight / wall B."""
    xb = C.curve_start_x(p, c) + p.curve_r
    return Pos(xb, 0, 0) * Rot(0, 0, 90)


def span_location(sp: C.Span) -> Location:
    hd = math.degrees(math.atan2(sp.b.y - sp.a.y, sp.b.x - sp.a.x))
    return Pos(sp.a.x, sp.a.y, 0) * Rot(0, 0, hd)


def frame_lines(p: Params, curved: bool) -> list[tuple[str, float, float, float]]:
    """(kind, y line, x start, length) for the two frames of a span, in the span frame."""
    g = span_geom(p, curved)
    out = []
    for sgn, side in ((1, "inner"), (-1, "outer")):
        yl = g.yc + sgn * p.bow_frame_y
        xa, xb = frame_x_range(g, yl)
        kind = side if curved else "straight"
        out.append((kind, yl, xa + p.portal_gap, xb - xa - 2 * p.portal_gap))
    return out


def rod(p: Params, c: C.Corner, u: float) -> Part:
    """Rod u to the left of the track, from P0 to the last portal."""
    r_disc = (p.rod_d) / 2
    x1, r = C.curve_start_x(p, c), p.curve_r
    parts = []
    if x1 > 0:
        parts.append(Pos(0, u, p.rod_zc) * extrude(Plane.YZ * Circle(r_disc), x1))
    face = Plane.XZ * (Pos(r - u, p.rod_zc) * Circle(r_disc))
    parts.append(Pos(x1, r, 0) * Rot(0, 0, -90) * revolve(face, Axis.Z, 90))
    ln = c.n_out * p.bow_span
    parts.append(Pos(x1 + r - u, r, p.rod_zc) * extrude(Plane.XZ * Circle(r_disc), -ln))
    out = parts[0]
    for q in parts[1:]:
        out += q
    return out


def placed_parts(p: Params = P, c: C.Corner = C.inside_corner()) -> list[tuple[str, Part]]:
    out = []
    decks = {cv: bow_deck.build(p, cv) for cv in (False, True)}
    for cv, d in decks.items():
        d.label = f"corner_deck_{'curve' if cv else 'straight'}"
    for sp in C.spans(p, c):
        loc = span_location(sp)
        out.append((f"deck{sp.index}", loc * decks[sp.curved]))
        for kind, yl, xs, ln in frame_lines(p, sp.curved):
            fr = bow_frame.make(p, ln)
            fr.label = f"corner_frame_{kind}"
            side = "inner" if yl > span_geom(p, sp.curved).yc else "outer"
            out.append((f"frame{sp.index}_{side}",
                        loc * Pos(xs, yl + p.bow_frame_t / 2, 0) * Rot(90, 0, 0) * fr))
    ps = C.portals(p, c)
    fb = floor_beam.make(p)
    fb.label = "floor_beam"
    for i, pt in enumerate(ps[1:-1], start=1):
        out.append((f"floor_beam{i}", floor_beam.place(p, pt.x, pt.y, pt.heading) * fb))
    anc = rod_anchor.make(p)
    anc.label = "rod_anchor"
    out.append(("anchor_in", rod_anchor.place(p, 0) * anc))
    out.append(("anchor_out", wall_b_frame(p, c) * rod_anchor.place(p, ps[-1].y) * anc))
    xb = C.curve_start_x(p, c) + p.curve_r + p.track_offset
    for i, wall in c.deep:
        pt = ps[i]
        if wall == "A":
            db = deep_bracket.make(p, pt.y + p.track_offset, pt.heading)
            db.label = "deep_bracket"
            out.append((f"deep{i}", deep_bracket.place(p, pt.x) * db))
        else:
            db = deep_bracket.make(p, xb - pt.x, 90 - pt.heading)
            db.label = "deep_bracket"
            out.append((f"deep{i}", wall_b_frame(p, c) * deep_bracket.place(p, pt.y) * db))
    for u, side in ((p.rod_y, "inner"), (-p.rod_y, "outer")):
        r = rod(p, c, u)
        r.label = f"aluminum_rod_{side}"
        out.append((f"rod_{side}", r))
    return out


def cables(p: Params, c: C.Corner) -> list[tuple[str, Part]]:
    """Ceiling-drop cables as thin cylinders from each floor-beam eye up to the ceiling."""
    out = []
    ps = C.portals(p, c)
    top = p.rail_top + p.rail_top_below_ceiling
    u = floor_beam.half_len(p) - 4
    z0 = floor_beam.bottom(p) / 2
    for i in c.ceiling:
        pt = ps[i]
        a = math.radians(pt.heading)
        for s in (1, -1):
            x, y = pt.x - s * u * math.sin(a), pt.y + s * u * math.cos(a)
            out.append((f"cable{i}_{s}", Pos(x, y, z0) * Cylinder(p.cable_d / 2, top - z0,
                                                                   align=(None, None, None)).moved(Location((0, 0, (top - z0) / 2)))))
    return out


def make(p: Params = P, c: C.Corner = C.inside_corner()) -> Compound:
    return Compound(children=[s for _, s in placed_parts(p, c)], label=f"{c.name}_corner")
```

```python
from hotrain.params import P, Params
from hotrain.parts import arch_half, bracket, cross_frame, end_pier, handrail, keystone, tie_ladder

GAP = 3.0  # mm between parts on a plate


@dataclass
class Plate:
    name: str
    qty: int
    items: list  # (part name, shape placed on the bed)


@dataclass
class _Rect:
    x: float
    y: float
    w: float
    h: float


def _zeroed(s: Shape) -> Shape:
    bb = s.bounding_box()
    return Pos(-bb.min.X, -bb.min.Y, -bb.min.Z) * s


@lru_cache(maxsize=None)
def nested_arch_pair(p: Params, bay_len: float) -> Compound:
    """Two arch halves head-to-tail (one turned 180°), as close as GAP allows."""
    a = _zeroed(arch_half.make(p, bay_len))
    w, h = a.bounding_box().size.X, a.bounding_box().size.Y
    low = _zeroed(Rot(0, 0, 180) * a)
    lo, hi = 0.0, h + GAP + 1
    for _ in range(14):  # binary search the smallest lift that keeps GAP clearance
        mid = (lo + hi) / 2
        if (Pos(0, mid, 0) * a).distance_to(low) >= GAP:
            hi = mid
        else:
            lo = mid
    return Compound(children=[low, Pos(0, hi, 0) * a])


def _free_after(free: list[_Rect], u: _Rect) -> list[_Rect]:
    out = []
    for r in free:
        if u.x >= r.x + r.w or u.x + u.w <= r.x or u.y >= r.y + r.h or u.y + u.h <= r.y:
            out.append(r)
            continue
        if u.x > r.x:
            out.append(_Rect(r.x, r.y, u.x - r.x, r.h))
        if u.x + u.w < r.x + r.w:
            out.append(_Rect(u.x + u.w, r.y, r.x + r.w - u.x - u.w, r.h))
        if u.y > r.y:
            out.append(_Rect(r.x, r.y, r.w, u.y - r.y))
        if u.y + u.h < r.y + r.h:
            out.append(_Rect(r.x, u.y + u.h, r.w, r.y + r.h - u.y - u.h))
    inside = lambda a, b: a.x >= b.x and a.y >= b.y and a.x + a.w <= b.x + b.w and a.y + a.h <= b.y + b.h
    return [a for i, a in enumerate(out) if not any(j != i and inside(a, b) for j, b in enumerate(out))]


def pack(p: Params, items: list[tuple[str, Shape]]) -> list[list[tuple[str, Shape]]]:
    """Pack (name, flat shape) items onto as few beds as possible. Compounds stay together."""
    side = p.bed - p.bed_margin + GAP  # every item carries GAP on its right/top edge
    off = p.bed_margin / 2
    todo = []
    for name, s in items:
        s = _zeroed(s)
        bb = s.bounding_box()
        todo.append((name, s, bb.size.X + GAP, bb.size.Y + GAP))
    todo.sort(key=lambda t: (-max(t[2], t[3]), -t[2] * t[3]))
    plates_out = []
    while todo:
        free, placed, left = [_Rect(0, 0, side, side)], [], []
        for name, s, w, h in todo:
            best = None
            for r in free:
                for ww, hh, rot in ((w, h, False), (h, w, True)):
                    if ww <= r.w + 1e-9 and hh <= r.h + 1e-9:
                        key = (min(r.w - ww, r.h - hh), max(r.w - ww, r.h - hh))
                        if best is None or key < best[0]:
                            best = (key, r.x, r.y, ww, hh, rot)
            if best is None:
                left.append((name, s, w, h))
                continue
            _, x, y, ww, hh, rot = best
            shape = _zeroed(Rot(0, 0, 90) * s) if rot else s
            placed.append((name, Pos(x + off, y + off, 0) * shape))
            free = _free_after(free, _Rect(x, y, ww, hh))
        if not placed:
            raise ValueError(f"{left[0][0]} does not fit on the bed")
        plates_out.append(placed)
        todo = left
    return plates_out


def _ladder_key(p: Params, bays: list[float], seg: L.Segment) -> tuple:
    r = lambda v: round(v - seg.x_start, 2)
    return ("ladder", round(seg.x_end - seg.x_start, 2), seg.female_start, seg.male_end,
            tuple((r(t.x), t.long, t.lip) for t in L.ties(p, bays) if seg.x_start < t.x < seg.x_end),
            tuple(r(x) for x in L.tab_xs(p, bays) if seg.x_start < x < seg.x_end))


def _rail_key(pc: L.HandrailPiece) -> tuple:
    return ("rail", round(pc.x_end - pc.x_start, 2), tuple(round(x - pc.x_start, 2) for x in pc.posts))


def _kits(p: Params, bays: list[float]) -> list[tuple[str, tuple, list]]:
    """(role, dedupe key, [(part name, shape-or-compound)]) per kit, in run order."""
    segs, rails = L.segments(p, bays), L.handrail_pieces(p, bays)
    rails_in = lambda s: [(k, pc) for k, pc in enumerate(rails) if s.x_start <= (pc.x_start + pc.x_end) / 2 < s.x_end]

    def deck(k: int) -> tuple[list, list]:
        s = segs[k]
        items = [(f"ladder_{k:02d}", tie_ladder.make(p, bays, s))]
        keys = [_ladder_key(p, bays, s)]
        for j, pc in rails_in(s):
            items.append((f"handrail_{j:02d}", handrail.make(p, pc)))
            keys.append(_rail_key(pc))
        return items, keys

    kits = []
    for i, b in enumerate(bays):
        tag = f"{b:.1f}mm"
        half = [(f"arch_half_{tag}#pair", nested_arch_pair(p, b)), (f"keystone", keystone.make(p))]
        half += [(f"cross_frame_{tag}_{k}", cross_frame.make(p, L.cf_depth(p, b, u)))
                 for k, u in enumerate(L.post_us(p, b))]
        for role, k, extra in (("A", 2 * i + 1, []), ("B", 2 * i + 2, [("bracket", bracket.make(p))])):
            d_items, d_keys = deck(k)
            key = (role, tag, tuple(d_keys))
            kits.append((f"bay_{tag}_{role}", key, half + extra + d_items))
    d_items, d_keys = deck(0)
    ends = [("end_pier", end_pier.make(p)) for _ in range(4)] + [("bracket", bracket.make(p))] + d_items
    kits.append(("run_ends", ("E", tuple(d_keys)), ends))
    return kits


def plates(p: Params = P, bays: list[float] = L.BAYS) -> list[Plate]:
    """Distinct plates with how many of each to print. Item names match out/manifest.csv
    (a nested arch pair counts as two arch halves)."""
    L.validate_bays(p, bays)
    seen: dict[tuple, list[Plate]] = {}
    variants: dict[str, int] = {}
    order: list[tuple] = []
    for role, key, items in _kits(p, bays):
        if key in seen:
            for pl in seen[key]:
                pl.qty += 1
            continue
        variants[role] = variants.get(role, 0) + 1
        base = role if variants[role] == 1 else f"{role}{variants[role]}"
        packed = pack(p, items)
        out = []
        for n, pl in enumerate(packed):
            flat = []
            for name, s in pl:
                if name.endswith("#pair"):
                    flat += [(name.replace("#pair", "#1"), s.solids()[0]), (name.replace("#pair", "#2"), s.solids()[1])]
                else:
                    flat.append((name, s))
            out.append(Plate(base if len(packed) == 1 else f"{base}_{n + 1}", 1, flat))
        seen[key] = out
        order.append(key)
    return [pl for key in order for pl in seen[key]]


def corner_kit(p: Params, corner) -> list[tuple[str, Shape]]:
    """Every printed part of one corner, named like out/corners/*.step."""
    from hotrain import corner_assembly as CA
    from hotrain.parts import bow_deck, bow_frame, deep_bracket, floor_beam, rod_anchor
    from hotrain import corner_layout as CL
    items = []
    for sp in CL.spans(p, corner):
        kind = "curve" if sp.curved else "straight"
        items.append((f"corner_deck_{kind}", bow_deck.make(p, sp.curved)))
        for k, _, _, ln in CA.frame_lines(p, sp.curved):
            items.append((f"corner_frame_{k}", bow_frame.make(p, ln)))
    n_beams = len(CL.portals(p, corner)) - 2
    items += [("floor_beam", floor_beam.make(p)) for _ in range(n_beams)]
    items += [("rod_anchor", rod_anchor.make(p)) for _ in range(2)]
    ps = CL.portals(p, corner)
    for i, wall in corner.deep:
        pt = ps[i]
        ang = pt.heading if wall == "A" else 90 - pt.heading
        items.append(("deep_bracket", deep_bracket.make(p, CL.wall_offset(p, corner, pt), ang)))
    return items


def corner_plates(p: Params = P) -> list[Plate]:
    """Plates for the three inside corners (printed 3x), the bridge corner, and the rod-bending jig."""
    from hotrain import corner_layout as CL
    from hotrain.parts import rod_jig
    out = []
    for corner, qty in ((CL.inside_corner(), 3), (CL.bridge_corner(), 1)):
        packed = pack(p, corner_kit(p, corner))
        for n, pl in enumerate(packed):
            name = f"{corner.name}_corner" + (f"_{n + 1}" if len(packed) > 1 else "")
            out.append(Plate(name, qty, pl))
    out.append(Plate("rod_jig", 1, pack(p, [("rod_jig", rod_jig.make(p))])[0]))
    return out
```

- [ ] **Step 5: Write `hotrain/build.py` (full file)**

```python
"""Command line: export STEP files (and optional STL / SVG previews) for the run in layout.BAYS.

    uv run python -m hotrain.build all          # parts + assembly + fit test
    uv run python -m hotrain.build parts        # one STEP per distinct part + out/manifest.csv
    uv run python -m hotrain.build assembly     # out/step/run_assembly.step + demo_two_bays.step
    uv run python -m hotrain.build fit-test     # out/step/fit_test.step
    uv run python -m hotrain.build plates       # out/plates/*.step + *.3mf, packed build plates + out/plates.csv
    uv run python -m hotrain.build corners      # out/corners/*.step, corner assemblies, out/corners_layout.csv
    uv run python -m hotrain.build check        # interference check, exit 1 on any clash
Options: --out DIR (default out), --mesh (also STL), --preview (also SVG line drawings)
"""
import argparse
import csv
import sys
from collections import Counter
from pathlib import Path

from build123d import Compound, ExportSVG, LineType, Mesher, Shape, export_step, export_stl

from hotrain import assembly, plates
from hotrain import layout as L
from hotrain.params import P, Params
from hotrain.parts import arch_half, bracket, cross_frame, end_pier, fit_test, handrail, keystone, tie_ladder

TOP = ((0, 0, 1), (0, 1, 0))  # (view direction toward the eye, up) for flat parts
FROM_ROOM = ((0, 1, 0), (0, 0, 1))  # assembly seen from the room
CORNER_VIEW = ((-0.6, -1.0, 0.7), (0, 0, 1))


def write_svg(shape: Shape, path: Path, view: tuple) -> None:
    direction, up = view
    bb = shape.bounding_box()
    c = bb.center()
    dist = 10 * max(bb.size.X, bb.size.Y, bb.size.Z)
    origin = (c.X + direction[0] * dist, c.Y + direction[1] * dist, c.Z + direction[2] * dist)
    visible, hidden = shape.project_to_viewport(origin, viewport_up=up, look_at=(c.X, c.Y, c.Z))
    span = max(bb.size.X, bb.size.Y, bb.size.Z)
    exp = ExportSVG(scale=800 / span)
    exp.add_layer("visible")
    exp.add_layer("hidden", line_color=(0.63, 0.63, 0.63), line_type=LineType.ISO_DOT)
    exp.add_shape(visible, layer="visible")
    exp.add_shape(hidden, layer="hidden")
    path.parent.mkdir(parents=True, exist_ok=True)
    exp.write(str(path))


def export(shape: Shape, out: Path, name: str, mesh: bool, preview: bool, view: tuple = TOP,
           subdir: str = "step") -> Path:
    step = out / subdir / f"{name}.step"
    step.parent.mkdir(parents=True, exist_ok=True)
    if not isinstance(shape, Compound) or "assembly" not in name and name != "demo_two_bays":
        shape.label = name
    export_step(shape, str(step))
    if mesh:
        stl = out / "stl" / f"{name}.stl"
        stl.parent.mkdir(parents=True, exist_ok=True)
        export_stl(shape, str(stl))
    if preview:
        write_svg(shape, out / "preview" / f"{name}.svg", view)
    return step


def part_list(p: Params, bays: list[float]) -> list[tuple[str, Shape, int, str]]:
    """(name, part, quantity, note) for every distinct printable part in the run."""
    rows = []
    for b, n in sorted(Counter(round(b, 3) for b in bays).items()):
        tag = f"{b:.1f}mm"
        rows.append((f"arch_half_{tag}", arch_half.make(p, b), 4 * n, f"{n} bay(s) of {b:.1f} mm; 2 panels x 2 halves"))
        for k, u in enumerate(L.post_us(p, b)):
            d = L.cf_depth(p, b, u)
            rows.append((f"cross_frame_{tag}_{k}", cross_frame.make(p, d), 2 * n, f"post {k + 1} from each pier"))
    rows.append(("keystone", keystone.make(p), 2 * len(bays), "one per panel per bay"))
    rows.append(("bracket", bracket.make(p), len(bays) + 1, "one per pier"))
    rows.append(("end_pier", end_pier.make(p), 4, "two per run end"))
    for k, seg in enumerate(L.segments(p, bays)):
        rows.append((f"ladder_{k:02d}", tie_ladder.make(p, bays, seg), 1,
                     f"x {seg.x_start:.1f} to {seg.x_end:.1f} mm"))
    for k, pc in enumerate(L.handrail_pieces(p, bays)):
        rows.append((f"handrail_{k:02d}", handrail.make(p, pc), 1,
                     f"x {pc.x_start:.1f} to {pc.x_end:.1f} mm, {len(pc.posts)} posts"))
    return rows


def cmd_parts(p: Params, bays: list[float], out: Path, mesh: bool, preview: bool) -> None:
    rows = part_list(p, bays)
    for name, shape, _, _ in rows:
        export(shape, out, name, mesh, preview)
    with open(out / "manifest.csv", "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["file", "quantity", "note"])
        for name, _, qty, note in rows:
            w.writerow([f"step/{name}.step", qty, note])


def cmd_assembly(p: Params, bays: list[float], out: Path, mesh: bool, preview: bool) -> None:
    export(assembly.make(p, bays), out, "run_assembly", mesh, preview, FROM_ROOM)
    # Two default bays sharing one bracket, to show how units join (independent of BAYS).
    export(assembly.make(p, [p.bay_len, p.bay_len]), out, "demo_two_bays", mesh, preview, FROM_ROOM)


def cmd_plates(p: Params, bays: list[float], out: Path, preview: bool) -> None:
    d = out / "plates"
    d.mkdir(parents=True, exist_ok=True)
    rows = []
    for pl in plates.plates(p, bays) + plates.corner_plates(p):
        children = []
        for name, s in pl.items:
            s.label = name
            children.append(s)
        comp = Compound(children=children, label=pl.name)
        export_step(comp, str(d / f"{pl.name}.step"))
        mesher = Mesher()
        for s in children:
            mesher.add_shape(s)
        mesher.write(str(d / f"{pl.name}.3mf"))
        if preview:
            write_svg(comp, out / "preview" / f"plate_{pl.name}.svg", TOP)
        counts = {}
        for name, _ in pl.items:
            base = name.split("#")[0]
            counts[base] = counts.get(base, 0) + 1
        rows.append([pl.name, pl.qty, "; ".join(f"{n} x{c}" for n, c in counts.items())])
    with open(out / "plates.csv", "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["plate", "quantity", "contents"])
        w.writerows(rows)


def cmd_corners(p: Params, out: Path, mesh: bool, preview: bool) -> None:
    from hotrain import corner_assembly, corner_layout
    from hotrain.parts import rod_jig
    seen = set()
    for corner in (corner_layout.inside_corner(), corner_layout.bridge_corner()):
        for name, shape in plates.corner_kit(p, corner):
            if name not in seen:
                seen.add(name)
                export(shape, out, name, mesh, preview, subdir="corners")
        export(corner_assembly.make(p, corner), out, f"{corner.name}_corner_assembly", mesh, preview,
               CORNER_VIEW, subdir="corners")
    export(rod_jig.make(p), out, "rod_jig", mesh, preview, subdir="corners")
    with open(out / "corners_layout.csv", "w", newline="") as f:
        w = csv.writer(f)
        w.writerow(["corner", "item", "measure", "mm", "inches"])
        for corner in (corner_layout.inside_corner(), corner_layout.bridge_corner()):
            for item, how, mm in corner_layout.layout_sheet(p, corner):
                w.writerow([corner.name, item, how, f"{mm:.1f}", f"{mm / 25.4:.2f}"])


def cmd_fit_test(p: Params, out: Path, mesh: bool, preview: bool) -> None:
    export(fit_test.make(p), out, "fit_test", mesh, preview)


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(prog="hotrain.build", description=__doc__.splitlines()[0])
    ap.add_argument("command", choices=["all", "parts", "assembly", "fit-test", "plates", "corners", "check"])
    ap.add_argument("--out", default="out")
    ap.add_argument("--mesh", action="store_true")
    ap.add_argument("--preview", action="store_true")
    a = ap.parse_args(argv)
    out, p, bays = Path(a.out), P, L.BAYS
    L.validate_bays(p, bays)
    out.mkdir(parents=True, exist_ok=True)
    if a.command == "check":
        clashes = assembly.find_clashes(assembly.placed_parts(p, bays))
        for na, nb, v in clashes:
            print(f"CLASH {na} x {nb}: {v} mm^3")
        print("no clashes" if not clashes else f"{len(clashes)} clash(es)")
        return 1 if clashes else 0
    if a.command in ("all", "parts"):
        cmd_parts(p, bays, out, a.mesh, a.preview)
    if a.command in ("all", "assembly"):
        cmd_assembly(p, bays, out, a.mesh, a.preview)
    if a.command in ("all", "fit-test"):
        cmd_fit_test(p, out, a.mesh, a.preview)
    if a.command in ("all", "corners"):
        cmd_corners(p, out, a.mesh, a.preview)
    if a.command in ("all", "plates"):
        cmd_plates(p, bays, out, a.preview)
    print(f"wrote {out}/")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
